type R = { tbl: string; code: string; name: string; norm: string };
function proposed(rows: R[], written: string, lookup: string[]) {
  const w = normName(written); const out: R[] = []; const seen = new Set<string>();
  const add = (r: R) => { const k = r.tbl + r.code; if (out.length < 20 && !seen.has(k)) { seen.add(k); out.push(r); } };
  const byCode = (a: R, b: R) => a.code.localeCompare(b.code);
  rows.filter((r) => r.norm === w).sort(byCode).forEach(add);
  const terms = lookup.map(normName).filter(Boolean);
  for (const t of terms) rows.filter((r) => r.norm === t).sort(byCode).forEach(add);
  const ring = [...terms, ...(w ? [w] : [])].map((t) => [...rows.filter((r) => r.norm.startsWith(t) && r.norm !== t).sort(byCode), ...rows.filter((r) => !r.norm.startsWith(t) && r.norm.includes(t)).sort(byCode)]);
  for (let i = 0; out.length < 20 && ring.some((l) => i < l.length); i++) for (const l of ring) if (i < l.length) add(l[i]);
  return out;
}
